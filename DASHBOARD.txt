let DATA_SELURUH_RENCANA_PTLK = [];

function aturPanelDaftarSeluruhRencanaPTLK() {
  // Panel pemantauan tersedia bagi seluruh pengguna yang berhasil masuk ke aplikasi.
  const panel = el('panelDaftarSeluruhRencana');
  if (!panel) return;
  panel.classList.remove('hidden');
  muatDaftarSeluruhRencanaPTLK();
}

function muatDaftarSeluruhRencanaPTLK() {
  status('statusDaftarSeluruhRencana', 'Sedang mengambil seluruh rencana kegiatan dari database...');
  el('tabelDaftarSeluruhRencana').innerHTML = '<tr><td colspan="8">Memuat data...</td></tr>';
  callServer('ambilDaftarRencanaKegiatanPTLK', [TOKEN_LOGIN, {}], hasil => {
    if (!hasil || !hasil.ok) {
      DATA_SELURUH_RENCANA_PTLK = [];
      status('statusDaftarSeluruhRencana', hasil && hasil.pesan ? hasil.pesan : 'Daftar seluruh rencana gagal dimuat.');
      el('tabelDaftarSeluruhRencana').innerHTML = '<tr><td colspan="8">Data gagal dimuat.</td></tr>';
      perbaruiRingkasanSeluruhRencanaPTLK([]);
      return;
    }
    DATA_SELURUH_RENCANA_PTLK = Array.isArray(hasil.data) ? hasil.data : [];
    isiFilterSeluruhRencanaPTLK();
    renderDaftarSeluruhRencanaPTLK();
    status('statusDaftarSeluruhRencana', 'Data database berhasil dimuat: ' + DATA_SELURUH_RENCANA_PTLK.length + ' rencana kegiatan. Gunakan filter untuk menyaring daftar.');
  }, m => {
    status('statusDaftarSeluruhRencana', 'Gagal mengambil daftar seluruh rencana: ' + m);
    el('tabelDaftarSeluruhRencana').innerHTML = '<tr><td colspan="8">Terjadi kesalahan saat mengambil data.</td></tr>';
  });
}

function isiFilterSeluruhRencanaPTLK() {
  const tahunLama = el('filterSemuaRencanaTahun').value;
  const wilayahLama = el('filterSemuaRencanaWilayah').value;
  const tahun = [...new Set(DATA_SELURUH_RENCANA_PTLK.map(x => String(x.tahun || '')).filter(Boolean))].sort((a,b) => Number(b)-Number(a));
  const wilayah = [...new Set(DATA_SELURUH_RENCANA_PTLK.map(x => String(x.kabupatenKota || '').trim()).filter(Boolean))].sort((a,b) => a.localeCompare(b, 'id'));
  setOptions('filterSemuaRencanaTahun', tahun, 'Semua tahun');
  setOptions('filterSemuaRencanaWilayah', wilayah, 'Semua kabupaten/kota');
  if (tahun.includes(tahunLama)) el('filterSemuaRencanaTahun').value = tahunLama;
  if (wilayah.includes(wilayahLama)) el('filterSemuaRencanaWilayah').value = wilayahLama;
}

function perbaruiRingkasanSeluruhRencanaPTLK(data) {
  const daftar = data || [];
  const hitung = status => daftar.filter(x => String(x.statusPersetujuan || '').trim().toUpperCase() === status).length;
  el('jumlahSemuaRencana').textContent = daftar.length;
  el('jumlahRencanaDraf').textContent = hitung('DRAF');
  el('jumlahRencanaDiajukan').textContent = hitung('DIAJUKAN');
  el('jumlahRencanaDisetujui').textContent = hitung('DISETUJUI');
  el('jumlahRencanaDitolak').textContent = hitung('DITOLAK');
}

function renderDaftarSeluruhRencanaPTLK() {
  const statusFilter = el('filterSemuaRencanaStatus').value.trim().toUpperCase();
  const tahunFilter = el('filterSemuaRencanaTahun').value;
  const wilayahFilter = el('filterSemuaRencanaWilayah').value.trim().toUpperCase();
  const kata = el('cariSemuaRencana').value.trim().toLowerCase();
  const semua = DATA_SELURUH_RENCANA_PTLK;
  perbaruiRingkasanSeluruhRencanaPTLK(semua);
  const daftar = semua.filter(x => {
    const st = String(x.statusPersetujuan || '').trim().toUpperCase();
    const th = String(x.tahun || '');
    const wilayah = String(x.kabupatenKota || '').trim().toUpperCase();
    const cari = (String(x.idKegiatan || '') + ' ' + String(x.namaKegiatan || '')).toLowerCase();
    return (!statusFilter || st === statusFilter) && (!tahunFilter || th === tahunFilter) && (!wilayahFilter || wilayah === wilayahFilter) && (!kata || cari.includes(kata));
  }).sort((a,b) => Number(b.tahun || 0)-Number(a.tahun || 0) || String(b.idKegiatan || '').localeCompare(String(a.idKegiatan || '')));
  const body = el('tabelDaftarSeluruhRencana');
  body.innerHTML = '';
  if (!daftar.length) {
    body.innerHTML = '<tr><td colspan="8" class="muted">Tidak ada rencana yang sesuai dengan filter.</td></tr>';
    status('statusDaftarSeluruhRencana', 'Tidak ada data yang sesuai filter. Total data dalam database: ' + semua.length + '.');
    return;
  }
  daftar.forEach(x => {
    const tr = document.createElement('tr');
    const values = [x.idKegiatan, x.tahun, x.namaKegiatan, x.kabupatenKota, Number(x.anggaranRencana || 0).toLocaleString('id-ID', {style:'currency',currency:'IDR',maximumFractionDigits:0})];
    values.forEach(v => { const td = document.createElement('td'); td.textContent = v == null || v === '' ? '-' : String(v); tr.appendChild(td); });
    const tdStatus = document.createElement('td');
    const badge = document.createElement('span');
    const st = String(x.statusPersetujuan || 'BELUM DIISI').trim().toUpperCase();
    badge.className = 'status ' + st.replace(/[^A-Z0-9_]/g, '');
    badge.textContent = st;
    tdStatus.appendChild(badge); tr.appendChild(tdStatus);
    const tdPelaksanaan = document.createElement('td'); tdPelaksanaan.textContent = x.statusPelaksanaan || '-'; tr.appendChild(tdPelaksanaan);
    const tdAksi = document.createElement('td');
    const btn = document.createElement('button'); btn.type = 'button'; btn.className = 'small secondary'; btn.textContent = 'Lihat Detail';
    btn.addEventListener('click', () => bukaDetailDaftarSeluruhRencana(x.idKegiatan, btn));
    tdAksi.appendChild(btn); tr.appendChild(tdAksi);
    body.appendChild(tr);
  });
  status('statusDaftarSeluruhRencana', 'Menampilkan ' + daftar.length + ' dari ' + semua.length + ' rencana kegiatan.');
}

function putuskanRencanaPTLK(keputusan) {
  if (!DETAIL_RENCANA_AKTIF || !DETAIL_RENCANA_AKTIF.idKegiatan) {
    alert('Pilih dan buka detail rencana terlebih dahulu.'); return;
  }
  const catatan = el('catatanPersetujuanRencana').value.trim();
  if (keputusan === 'DITOLAK' && !catatan) {
    alert('Alasan penolakan wajib diisi.'); el('catatanPersetujuanRencana').focus(); return;
  }
  const label = keputusan === 'DISETUJUI' ? 'MENYETUJUI' : 'MENOLAK';
  if (!confirm('Apakah Anda yakin akan ' + label + ' rencana kegiatan ini? Keputusan akan langsung disimpan ke database.')) return;
  const id = DETAIL_RENCANA_AKTIF.idKegiatan;
  const tombolSetuju = el('btnSetujuiRencana'), tombolTolak = el('btnTolakRencana');
  tombolSetuju.disabled = true; tombolTolak.disabled = true;
  status('statusPersetujuanRencana', 'Sedang menyimpan keputusan ' + keputusan + '...');
  callServer('verifikasiRencanaKegiatanPTLK', [TOKEN_LOGIN, id, keputusan, catatan], hasil => {
    if (!hasil || !hasil.ok) {
      tombolSetuju.disabled = false; tombolTolak.disabled = false;
      status('statusPersetujuanRencana', hasil && hasil.pesan ? hasil.pesan : 'Keputusan gagal disimpan.');
      return;
    }
    pesan(hasil.pesan || 'Keputusan berhasil disimpan.', 'ok');
    tutupDetailPersetujuanRencana();
    muatDaftarPersetujuanRencana();
    muatDaftarSeluruhRencanaPTLK();
  }, m => {
    tombolSetuju.disabled = false; tombolTolak.disabled = false;
    status('statusPersetujuanRencana', 'Gagal menyimpan keputusan: ' + m);
  });
}

function aturPanelRencanaDitolak() {
  const panel = el('panelRencanaDitolak');
  if (!panel) return;
  const peran = String(PERAN_PENGGUNA || '').trim().toUpperCase();
  const boleh = ['ADMIN_UTAMA', 'OPERATOR_PENGELOLA_DATA', 'OPERATOR_PENGOLAH_DATA'].includes(peran);
  panel.classList.toggle('hidden', !boleh);
  if (boleh) muatDaftarRencanaDitolak();
}

function muatDaftarRencanaDitolak() {
  const peran = String(PERAN_PENGGUNA || '').trim().toUpperCase();
  if (!['ADMIN_UTAMA', 'OPERATOR_PENGELOLA_DATA', 'OPERATOR_PENGOLAH_DATA'].includes(peran)) {
    status('statusRencanaDitolak', 'Akses hanya untuk Admin Utama dan Operator Pengelola/Pengolah Data.');
    return;
  }
  status('statusRencanaDitolak', 'Sedang mengambil rencana berstatus DITOLAK...');
  el('tabelRencanaDitolak').innerHTML = '<tr><td colspan="7">Memuat data...</td></tr>';
  callServer('ambilDaftarRencanaKegiatanPTLK', [TOKEN_LOGIN, {}], hasil => {
    if (!hasil || !hasil.ok) {
      DATA_RENCANA_DITOLAK = [];
      status('statusRencanaDitolak', hasil && hasil.pesan ? hasil.pesan : 'Daftar rencana ditolak gagal dimuat.');
      el('tabelRencanaDitolak').innerHTML = '<tr><td colspan="7">Data gagal dimuat.</td></tr>';
      return;
    }
    DATA_RENCANA_DITOLAK = (hasil.data || []).filter(x => String(x.statusPersetujuan || '').trim().toUpperCase() === 'DITOLAK');
    status('statusRencanaDitolak', 'Jumlah rencana yang perlu diperbaiki: ' + DATA_RENCANA_DITOLAK.length);
    renderDaftarRencanaDitolak();
  }, m => {
    status('statusRencanaDitolak', 'Gagal mengambil daftar: ' + m);
    el('tabelRencanaDitolak').innerHTML = '<tr><td colspan="7">Terjadi kesalahan saat mengambil data.</td></tr>';
  });
}

function renderDaftarRencanaDitolak() {
  const body = el('tabelRencanaDitolak');
  body.innerHTML = '';
  if (!DATA_RENCANA_DITOLAK.length) {
    body.innerHTML = '<tr><td colspan="7" class="muted">Belum ada rencana kegiatan berstatus DITOLAK.</td></tr>';
    return;
  }
  DATA_RENCANA_DITOLAK.forEach(x => {
    const tr = document.createElement('tr');
    [x.idKegiatan, x.tahun, x.namaKegiatan, x.kabupatenKota, Number(x.anggaranRencana || 0).toLocaleString('id-ID', {style:'currency',currency:'IDR',maximumFractionDigits:0}), x.statusPersetujuan].forEach(v => {
      const td = document.createElement('td'); td.textContent = v == null || v === '' ? '-' : String(v); tr.appendChild(td);
    });
    const td = document.createElement('td');
    const btn = document.createElement('button'); btn.type = 'button'; btn.className = 'small'; btn.textContent = 'Buka dan Perbaiki';
    btn.addEventListener('click', () => bukaEditorRencanaDitolak(x.idKegiatan, btn));
    td.appendChild(btn); tr.appendChild(td); body.appendChild(tr);
  });
}

  const wilayah = prompt(

    wilayahDiajukan
  );

function tolakPengajuanOperator(id) {
  const catatan = prompt('Tuliskan alasan penolakan:');
  if (catatan === null) return;

  if (!catatan.trim()) {
    alert('Alasan penolakan wajib diisi.');
    return;
  }

  if (!confirm('Tolak pengajuan operator ini?')) return;

  prosesKeputusanPengajuanOperator(
    id, 'DITOLAK', '', catatan
  );
}

function prosesKeputusanPengajuanOperator(
  id, keputusan, wilayah, catatan
) {
  status('statusPengajuanOperator', 'Memproses keputusan Admin...');

  callServer(
    'prosesPengajuanOperatorKabKotaPTLK',
    [TOKEN_LOGIN, id, keputusan, wilayah, catatan],
    hasil => {
      if (!hasil || !hasil.ok) {
        status(
          'statusPengajuanOperator',
          hasil && hasil.pesan || 'Keputusan gagal diproses.'
        );

function pratinjauImporPTLK(jenis,inputId,areaId){
  const f=el(inputId),a=el(areaId);
  if(!f||!f.files||!f.files.length){alert('Pilih file Excel terlebih dahulu.');return;}
  const file=f.files[0];
  if(!/\.xlsx?$/i.test(file.name)){alert('File harus berformat .xlsx atau .xls.');return;}
  if(file.size>10*1024*1024){alert('Ukuran file maksimal 10 MB.');return;}
  if(typeof XLSX==='undefined'){a.textContent='Pustaka pembaca Excel belum termuat. Muat ulang aplikasi dan coba kembali.';a.className='message show error';return;}
  a.textContent='Membaca file Excel dan memeriksa data...';a.className='message show';
  const reader=new FileReader();
  reader.onload=function(e){
    try{
      const workbook=XLSX.read(e.target.result,{type:'array',cellDates:true});
      if(!workbook.SheetNames.length)throw new Error('Workbook tidak memiliki sheet.');
      const worksheet=workbook.Sheets[workbook.SheetNames[0]];
      const rows=XLSX.utils.sheet_to_json(worksheet,{defval:'',raw:false,blankrows:false});
      if(!rows.length)throw new Error('Sheet pertama tidak memiliki baris data setelah header.');
      callServer('pratinjauImporPTLK',[TOKEN_LOGIN,jenis,file.name,rows],function(h){
        if(!h||!h.ok){a.textContent=h&&h.pesan||'Pratinjau gagal.';a.className='message show error';return;}
        IMPOR_PTLK_AKTIF[jenis]=h.batchId;
        const r=h.ringkasan||{};
        a.replaceChildren();
        const ring=document.createElement('div');
        ring.textContent='Hasil pemeriksaan — Total: '+(r.total||0)+' | Valid: '+(r.valid||0)+' | Duplikat: '+(r.duplikat||0)+' | Tidak valid: '+(r.tidakValid||0);
        a.appendChild(ring);
        const wrap=document.createElement('div');wrap.className='table-wrap';wrap.style.marginTop='10px';
        const table=document.createElement('table');
        const thead=document.createElement('thead'),trh=document.createElement('tr');
        ['Baris Excel','Status','Nama/Identitas','Catatan pemeriksaan'].forEach(t=>{const th=document.createElement('th');th.textContent=t;trh.appendChild(th);});
        thead.appendChild(trh);table.appendChild(thead);
        const tbody=document.createElement('tbody');
        (h.baris||[]).forEach(x=>{const tr=document.createElement('tr');[x.baris,x.status,x.nama,x.catatan].forEach(v=>{const td=document.createElement('td');td.textContent=String(v==null?'':v);tr.appendChild(td);});tbody.appendChild(tr);});
        table.appendChild(tbody);wrap.appendChild(table);a.appendChild(wrap);
        const note=document.createElement('p');note.textContent=h.pesan||'Hanya baris VALID yang dapat disimpan.';a.appendChild(note);
        if((r.valid||0)>0&&h.batchId){const btn=document.createElement('button');btn.className='success small';btn.type='button';btn.textContent='Konfirmasi Simpan '+r.valid+' Baris';btn.onclick=()=>konfirmasiImporPTLK(jenis,areaId,btn);a.appendChild(btn);}
        a.className='message show ok';
      },function(m){a.textContent='Pemeriksaan gagal: '+m;a.className='message show error';});
    }catch(err){a.textContent='File Excel tidak dapat dibaca: '+err.message;a.className='message show error';}
  };
  reader.onerror=function(){a.textContent='File tidak dapat dibaca.';a.className='message show error';};
  reader.readAsArrayBuffer(file);
}

    nomorSK: el('penetapanNomorSK').value.trim(),
    pejabatPenetap: el('penetapanPejabat').value.trim(),
    nilaiDitetapkan: el('penetapanNilai').value,
    tautanSK: el('penetapanTautanSK').value.trim(),
    catatan: el('penetapanCatatan').value.trim()
  };

  if (!data.idHibah || !data.nomorSK ||
      !data.pejabatPenetap || !data.nilaiDitetapkan) {
    status(
      'statusPenetapanHibah',
      'Lengkapi ID Hibah, nomor SK, pejabat penetap, dan nilai bantuan.'
    );
    return;
  }

  status('statusPenetapanHibah', 'Sedang menyimpan penetapan...');

  google.script.run
    .withSuccessHandler(function(res) {
      if (!res || !res.ok) {
        status(
          'statusPenetapanHibah',
          res ? res.pesan : 'Penetapan gagal disimpan.'
        );
        return;
      }

      status(
        'statusPenetapanHibah',
        res.pesan + ' ID: ' + res.idPenetapan
      );

      el('penetapanNomorSK').value = '';
      el('penetapanPejabat').value = '';
      el('penetapanNilai').value = '';
      el('penetapanTautanSK').value = '';
      el('penetapanCatatan').value = '';

      muatPenetapanHibahDashboardPTLK();
    })
    .withFailureHandler(function(err) {
      status(
        'statusPenetapanHibah',
        'Kesalahan: ' + err.message
      );
    })
    .simpanPenetapanHibahPTLK(TOKEN_LOGIN, data);
}

function muatPenetapanHibahDashboardPTLK() {
  const tbody = el('daftarPenetapanHibah');
  if (!tbody) return;

  tbody.innerHTML =
    '<tr><td colspan="8">Memuat data...</td></tr>';

  google.script.run
    .withSuccessHandler(function(res) {
      if (!res || !res.ok) {
        tbody.innerHTML =
          '<tr><td colspan="8">Gagal memuat data penetapan.</td></tr>';

        status(
          'statusPenetapanHibah',
          res ? res.pesan : 'Gagal membaca data.'
        );
        return;
      }

      if (!res.data || res.data.length === 0) {
        tbody.innerHTML =
          '<tr><td colspan="8">Belum ada data penetapan.</td></tr>';
        return;
      }

      tbody.innerHTML = '';

      res.data.forEach(function(item, index) {
        const tr = document.createElement('tr');

        const tanggal = item.tanggalPenetapan
          ? new Date(item.tanggalPenetapan).toLocaleDateString('id-ID')
          : '-';

        const nilai = Number(item.nilaiDitetapkan || 0)
          .toLocaleString('id-ID');

        const kolom = [
          index + 1,
          item.idHibah || '-',
          tanggal,
          item.nomorSK || '-',
          item.pejabatPenetap || '-',
          'Rp ' + nilai,
          item.statusPenetapan || '-'
        ];

        kolom.forEach(function(teks) {
          const td = document.createElement('td');
          td.textContent = teks;
          tr.appendChild(td);
        });

        const tdDokumen = document.createElement('td');

        if (item.tautanSK) {
          const link = document.createElement('a');
          link.href = item.tautanSK;
          link.target = '_blank';
          link.rel = 'noopener noreferrer';
          link.textContent = 'Lihat SK';
          tdDokumen.appendChild(link);
        } else {
          tdDokumen.textContent = '-';
        }

        tr.appendChild(tdDokumen);
        tbody.appendChild(tr);
      });
    })
    .withFailureHandler(function(err) {
      tbody.innerHTML =
        '<tr><td colspan="8">Kesalahan: ' +
        err.message + '</td></tr>';
    })
    .ambilPenetapanHibahPTLK(TOKEN_LOGIN, {});
}

  pilih.innerHTML = '<option value="">Pilih peserta</option>';

function PTLK_PESERTA_SIMPAN() {
  if (!PTLK_PESERTA_AKSES()) {
    alert('Akses modul tidak diizinkan.');
    return;
  }

  const idKegiatan = el('pesertaPilihKegiatan').value;
  const jenis = el('pesertaJenis').value;
  const idSumber = el('pesertaPilihSumber').value;
  const peran = el('pesertaPeran').value.trim();

  if (!idKegiatan) return alert('Pilih kegiatan terlebih dahulu.');
  if (!idSumber) return alert('Pilih tenaga atau lembaga.');
  if (!peran) return alert('Isi peran peserta.');

  const daftar = jenis === 'TENAGA'
    ? PTLK_PESERTA_TENAGA
    : PTLK_PESERTA_LEMBAGA;

  const sumber = daftar.find(x =>
    String(jenis === 'TENAGA' ? x.ID_TENAGA : x.ID_LEMBAGA) === String(idSumber)
  );

  if (!sumber) return alert('Data peserta tidak ditemukan.');

  const data = {
    idKegiatan: idKegiatan,
    jenisPeserta: jenis,
    idTenaga: jenis === 'TENAGA' ? idSumber : '',
    idLembaga: jenis === 'LEMBAGA' ? idSumber : '',
    namaPeserta: jenis === 'TENAGA'
      ? sumber.NAMA_LENGKAP
      : sumber.NAMA_LEMBAGA,
    asalLembaga: jenis === 'TENAGA'
      ? (sumber.ORGANISASI_KELEMBAGAAN || '')
      : (sumber.KABUPATEN_KOTA || ''),
    peranPeserta: peran,
    keterangan: el('pesertaKeterangan').value.trim()
  };

  status('statusPesertaKegiatan', 'Menyimpan peserta...');

  callServer(
    'simpanPesertaKegiatanPTLK',
    [TOKEN_LOGIN, data],
    hasil => {
      if (!hasil || !hasil.ok) {
        status(
          'statusPesertaKegiatan',
          hasil && hasil.pesan || 'Peserta gagal disimpan.'
        );
        return;
      }

      el('pesertaPeran').value = '';
      el('pesertaKeterangan').value = '';

      status('statusPesertaKegiatan', hasil.pesan || 'Peserta berhasil ditambahkan.');
      PTLK_PESERTA_MUAT_DAFTAR();
    },
    m => status('statusPesertaKegiatan', 'Gagal menyimpan peserta: ' + m)
  );
}

function PTLK_PESERTA_MUAT_DAFTAR() {
  const idKegiatan = el('pesertaPilihKegiatan').value;
  const body = el('tabelPesertaKegiatan');

  if (!idKegiatan) {
    body.innerHTML = '<tr><td colspan="7">Pilih kegiatan terlebih dahulu.</td></tr>';
    return;
  }

  status('statusPesertaKegiatan', 'Membaca daftar peserta...');

  callServer(
    'ambilPesertaKegiatanPTLK',
    [TOKEN_LOGIN, idKegiatan],
    hasil => {
      if (!hasil || !hasil.ok) {
        body.innerHTML = '<tr><td colspan="7">Gagal membaca peserta.</td></tr>';
        status('statusPesertaKegiatan', hasil && hasil.pesan || 'Gagal membaca peserta.');
        return;
      }

      PTLK_PESERTA_DAFTAR = hasil.data || [];
      PTLK_PESERTA_RENDER();

      status(
        'statusPesertaKegiatan',
        'Jumlah peserta terdaftar: ' + PTLK_PESERTA_DAFTAR.length
      );
    },
    m => status('statusPesertaKegiatan', 'Gagal membaca peserta: ' + m)
  );
}

function PTLK_PESERTA_RENDER() {
  const body = el('tabelPesertaKegiatan');
  body.innerHTML = '';

  if (!PTLK_PESERTA_DAFTAR.length) {
    body.innerHTML = '<tr><td colspan="7">Belum ada peserta terdaftar.</td></tr>';
    return;
  }

  PTLK_PESERTA_DAFTAR.forEach(x => {
    const tr = document.createElement('tr');

    function sel(value) {
      const td = document.createElement('td');
      td.textContent = value || '-';
      tr.appendChild(td);
    }

    sel(x.namaPeserta);
    sel(x.jenisPeserta);
    sel(x.asalLembaga);
    sel(x.peranPeserta);

    const tdStatus = document.createElement('td');
    const pilihStatus = document.createElement('select');

    PTLK_PESERTA_STATUS.forEach(s => {
      const op = document.createElement('option');
      op.value = s;
      op.textContent = s;
      pilihStatus.appendChild(op);
    });

    pilihStatus.value = x.statusKehadiran || 'BELUM DIKONFIRMASI';
    tdStatus.appendChild(pilihStatus);
    tr.appendChild(tdStatus);

    const tdKet = document.createElement('td');
    const inputKet = document.createElement('input');
    inputKet.value = x.keterangan || '';
    inputKet.placeholder = 'Keterangan';
    tdKet.appendChild(inputKet);
    tr.appendChild(tdKet);

    const tdAksi = document.createElement('td');
    const tombol = document.createElement('button');
    tombol.type = 'button';
    tombol.className = 'small success';
    tombol.textContent = 'Simpan';

    tombol.addEventListener('click', () => {
      tombol.disabled = true;

      callServer(
        'ubahKehadiranPesertaPTLK',
        [
          TOKEN_LOGIN,
          x.idPartisipasi,
          pilihStatus.value,
          inputKet.value.trim()
        ],
        hasil => {
          tombol.disabled = false;

          if (!hasil || !hasil.ok) {
            alert(hasil && hasil.pesan || 'Kehadiran gagal diperbarui.');
            return;
          }

          status('statusPesertaKegiatan', hasil.pesan || 'Kehadiran berhasil diperbarui.');
          PTLK_PESERTA_MUAT_DAFTAR();
        },
        m => {
          tombol.disabled = false;
          alert('Gagal memperbarui kehadiran: ' + m);
        }
      );
    });

    tdAksi.appendChild(tombol);
    tr.appendChild(tdAksi);
    body.appendChild(tr);
  });
}

function PTLK_LPJ_ESCAPE_(nilai) {
  return String(nilai == null ? '' : nilai)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

// MEMUAT DAFTAR LPJ
function muatLPJHibahDashboardPTLK() {
  status('statusLPJHibah', 'Sedang memuat daftar LPJ...');

  callServer(
    'ambilLPJHibahPTLK',
    [TOKEN_LOGIN, {}],
    hasil => {
      if (!hasil || hasil.ok === false) {
        status(
          'statusLPJHibah',
          hasil && hasil.pesan || 'Gagal memuat daftar LPJ.'
        );
        return;
      }

      const daftar = Array.isArray(hasil)
        ? hasil
        : (Array.isArray(hasil.data) ? hasil.data : []);

      const tbody = el('daftarLPJHibah');

      if (!daftar.length) {
        tbody.innerHTML =
          '<tr><td colspan="7">Belum ada data LPJ.</td></tr>';
        el('verifikasiLPJId').innerHTML =
          '<option value="">Belum ada LPJ</option>';
        status('statusLPJHibah', 'Belum ada data LPJ.');
        return;
      }
     
      tbody.innerHTML = daftar.map((item, index) => {
        const tautan = item.tautanLPJ || item.TAUTAN_LPJ || '';
        const statusLPJ = item.statusLPJ || item.STATUS_LPJ || '';

        return '<tr>' +
          '<td>' + (index + 1) + '</td>' +
          '<td>' + PTLK_LPJ_ESCAPE_(
            item.idHibah || item.ID_HIBAH
          ) + '</td>' +
          '<td>' + PTLK_LPJ_ESCAPE_(
            item.nomorLPJ || item.NOMOR_LPJ
          ) + '</td>' +
          '<td>' + PTLK_LPJ_ESCAPE_(
            item.tanggalPenyerahan || item.TANGGAL_PENYERAHAN
          ) + '</td>' +
          '<td>Rp' + Number(
            item.nilaiPertanggungjawaban ||
            item.NILAI_PERTANGGUNGJAWABAN || 0
          ).toLocaleString('id-ID') + '</td>' +
          '<td>' + PTLK_LPJ_ESCAPE_(statusLPJ) + '</td>' +
          '<td>' + (tautan
            ? '<a href="' + PTLK_LPJ_ESCAPE_(tautan) +
              '" target="_blank" rel="noopener">Buka LPJ</a>'
            : '-') + '</td>' +
          '</tr>';
      }).join('');

      const pilihan = '<option value="">Pilih LPJ</option>' +
        daftar.map(item => {
          const id = item.idLPJ || item.ID_LPJ || '';
          const nomor = item.nomorLPJ || item.NOMOR_LPJ || '';
          const statusLPJ = item.statusLPJ || item.STATUS_LPJ || '';

          if (!id) return '';

          return '<option value="' +
            PTLK_LPJ_ESCAPE_(id) + '">' +
            PTLK_LPJ_ESCAPE_(
              nomor + ' - ' + statusLPJ
            ) + '</option>';
        }).join('');

      el('verifikasiLPJId').innerHTML = pilihan;

      status(
        'statusLPJHibah',
        'Daftar LPJ berhasil dimuat: ' + daftar.length + ' data.'
      );
    },
    error => status(
      'statusLPJHibah',
      'Gagal memuat daftar LPJ: ' + error
    )
  );
}

// 1. MENYIMPAN HASIL MONITORING DAN EVALUASI HIBAH
function simpanMonevHibahPTLK(idToken, data) {
  const pengguna = PTLK_PERIKSA_HAK_HIBAH_();
  const lock = LockService.getScriptLock();

  try {
    if (!data) {
      throw new Error('Data monitoring tidak ditemukan.');
    }

    const idHibah = String(data.idHibah || '').trim();
    const tanggal = String(data.tanggalMonev || '').trim();
    const lokasi = String(data.lokasiMonev || '').trim();
    const petugas = String(data.namaPetugas || '').trim();
    const kesesuaian = String(
      data.kesesuaianPenggunaan || ''
    ).trim();

    const hasil = String(data.hasilMonev || '').trim();
    const manfaat = String(data.manfaatHibah || '').trim();
    const kendala = String(data.kendala || '').trim();
    const rekomendasi = String(data.rekomendasi || '').trim();
    const statusTindakLanjut = String(
      data.statusTindakLanjut || ''
    ).trim();

    if (!idHibah || !tanggal || !lokasi || !petugas) {
      throw new Error(
        'Penerima hibah, tanggal, lokasi, dan nama petugas wajib diisi.'
      );
    }

    if (!kesesuaian || !hasil || !manfaat) {
      throw new Error(
        'Kesesuaian penggunaan, hasil monitoring, dan manfaat hibah wajib diisi.'
      );
    }

    if (!statusTindakLanjut) {
      throw new Error('Status tindak lanjut wajib dipilih.');
    }

    if (!PTLK_HIBAH_ID_ADA_(idHibah)) {
      throw new Error(
        'ID Hibah tidak ditemukan dalam database Master Hibah.'
      );
    }

    if (!/^\d{4}-\d{2}-\d{2}$/.test(tanggal)) {
      throw new Error('Format tanggal monitoring tidak valid.');
    }

    const tanggalObj = new Date(
      Number(tanggal.substring(0, 4)),
      Number(tanggal.substring(5, 7)) - 1,
      Number(tanggal.substring(8, 10))
    );

    if (
      tanggalObj.getFullYear() !== Number(tanggal.substring(0, 4)) ||
      tanggalObj.getMonth() !== Number(tanggal.substring(5, 7)) - 1 ||
      tanggalObj.getDate() !== Number(tanggal.substring(8, 10))
    ) {
      throw new Error('Tanggal monitoring tidak valid.');
    }

    if (
      ['SESUAI', 'SEBAGIAN_SESUAI', 'TIDAK_SESUAI']
        .indexOf(kesesuaian) === -1
    ) {
      throw new Error('Pilihan kesesuaian penggunaan tidak valid.');
    }

    if (
      [
        'BELUM_DITINDAKLANJUTI',
        'DALAM_PROSES',
        'SELESAI'
      ].indexOf(statusTindakLanjut) === -1
    ) {
      throw new Error('Status tindak lanjut tidak valid.');
    }

    lock.waitLock(30000);

    const ss = ambilSpreadsheetPTLK_();
    const sheet = ss.getSheetByName('MONEV_HIBAH_PTLK');

    if (!sheet) {
      throw new Error('Sheet MONEV_HIBAH_PTLK tidak ditemukan.');
    }

    const jumlahKolom = sheet.getLastColumn();

    if (jumlahKolom < 1) {
      throw new Error('Header database Monev Hibah belum tersedia.');
    }

    const header = sheet
      .getRange(1, 1, 1, jumlahKolom)
      .getDisplayValues()[0]
      .map(function(x) {
        return String(x).trim().toUpperCase();
      });

    const kolomWajib = [
      'ID_MONEV',
      'ID_HIBAH',
      'TANGGAL_MONEV',
      'LOKASI_MONEV',
      'NAMA_PETUGAS',
      'KESESUAIAN_PENGGUNAAN',
      'HASIL_MONEV',
      'MANFAAT_HIBAH',
      'KENDALA',
      'REKOMENDASI',
      'STATUS_TINDAK_LANJUT'
    ];

    kolomWajib.forEach(function(namaKolom) {
      if (header.indexOf(namaKolom) === -1) {
        throw new Error(
          'Kolom database tidak ditemukan: ' + namaKolom
        );
      }
    });

    const nilai = {
      ID_MONEV: PTLK_ID_BARU_('MNV'),
      ID_HIBAH: idHibah,
      TANGGAL_MONEV: tanggalObj,
      LOKASI_MONEV: lokasi,
      NAMA_PETUGAS: petugas,
      KESESUAIAN_PENGGUNAAN: kesesuaian,
      HASIL_MONEV: hasil,
      MANFAAT_HIBAH: manfaat,
      KENDALA: kendala,
      REKOMENDASI: rekomendasi,
      STATUS_TINDAK_LANJUT: statusTindakLanjut
    };

    const baris = header.map(function(namaKolom) {
      return Object.prototype.hasOwnProperty.call(
        nilai,
        namaKolom
      ) ? nilai[namaKolom] : '';
    });

    sheet.appendRow(baris);

    return {
      ok: true,
      idMonev: nilai.ID_MONEV,
      pesan: 'Hasil Monitoring dan Evaluasi Hibah berhasil disimpan.'
    };

  } catch (error) {
    return {
      ok: false,
      pesan: error.message || String(error)
    };

  } finally {
    if (lock.hasLock()) {
      lock.releaseLock();
    }
  }
}


// 2. MEMBACA DAFTAR MONITORING DAN EVALUASI HIBAH
function ambilMonevHibahPTLK(idToken, filter) {
  try {
    PTLK_PERIKSA_HAK_HIBAH_(idToken);

    filter = filter || {};

    const ss = ambilSpreadsheetPTLK_();
    const sheet = ss.getSheetByName('MONEV_HIBAH_PTLK');

    if (!sheet) {
      throw new Error('Sheet MONEV_HIBAH_PTLK tidak ditemukan.');
    }

    const data = sheet.getDataRange().getValues();

    if (data.length < 2) {
      return {
        ok: true,
        data: [],
        jumlah: 0,
        pesan: 'Belum ada data Monitoring dan Evaluasi Hibah.'
      };
    }

    const header = data[0].map(function(x) {
      return String(x).trim().toUpperCase();
    });

    const kolom = {};
    header.forEach(function(nama, index) {
      kolom[nama] = index;
    });

    const wajib = [
      'ID_MONEV',
      'ID_HIBAH',
      'TANGGAL_MONEV',
      'LOKASI_MONEV',
      'NAMA_PETUGAS',
      'KESESUAIAN_PENGGUNAAN',
      'HASIL_MONEV',
      'MANFAAT_HIBAH',
      'KENDALA',
      'REKOMENDASI',
      'STATUS_TINDAK_LANJUT'
    ];

    wajib.forEach(function(nama) {
      if (kolom[nama] === undefined) {
        throw new Error(
          'Kolom database tidak ditemukan: ' + nama
        );
      }
    });

    const zona = Session.getScriptTimeZone();

    const daftar = data.slice(1)
      .filter(function(baris) {
        return String(baris[kolom.ID_MONEV] || '').trim() !== '';
      })
      .map(function(baris) {
        const tanggal = baris[kolom.TANGGAL_MONEV];

        return {
          idMonev: String(baris[kolom.ID_MONEV] || ''),
          idHibah: String(baris[kolom.ID_HIBAH] || ''),
          tanggalMonev: tanggal instanceof Date
            ? Utilities.formatDate(tanggal, zona, 'yyyy-MM-dd')
            : String(tanggal || ''),
          lokasiMonev: String(baris[kolom.LOKASI_MONEV] || ''),
          namaPetugas: String(baris[kolom.NAMA_PETUGAS] || ''),
          kesesuaianPenggunaan: String(
            baris[kolom.KESESUAIAN_PENGGUNAAN] || ''
          ),
          hasilMonev: String(baris[kolom.HASIL_MONEV] || ''),
          manfaatHibah: String(baris[kolom.MANFAAT_HIBAH] || ''),
          kendala: String(baris[kolom.KENDALA] || ''),
          rekomendasi: String(baris[kolom.REKOMENDASI] || ''),
          statusTindakLanjut: String(
            baris[kolom.STATUS_TINDAK_LANJUT] || ''
          )
        };
      });

    const hasil = daftar.filter(function(item) {
      if (
        filter.idHibah &&
        item.idHibah !== String(filter.idHibah)
      ) {
        return false;
      }

      return true;
    });

    hasil.reverse();

    return {
      ok: true,
      data: hasil,
      jumlah: hasil.length,
      pesan: 'Data Monitoring dan Evaluasi Hibah berhasil dibaca.'
    };

  } catch (error) {
    return {
      ok: false,
      data: [],
      jumlah: 0,
      pesan: error.message || String(error)
    };
  }
}

let DATA_KEGIATAN_MONEV_PTLK = [];

function PTLK_MONEV_ESCAPE_HTML_(nilai) {
  return String(nilai == null ? '' : nilai)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

function bukaSubPanelMonevPTLK(nama) {
  const daftar = [
    'monitoring',
    'tindakLanjut',
    'evaluasi',
    'dokumentasi'
  ];

  const idPanel = {
    monitoring: 'subMonevMonitoring',
    tindakLanjut: 'subMonevTindakLanjut',
    evaluasi: 'subMonevEvaluasi',
    dokumentasi: 'subMonevDokumentasi'
  };

  daftar.forEach(function(item) {
    const panel = el(idPanel[item]);

    if (panel) {
      panel.classList.toggle('hidden', item !== nama);
    }
  });

  if (!PTLK_MONEV_ID_KEGIATAN_TERPILIH_()) {
    alert('Pilih kegiatan terlebih dahulu.');
    return;
  }

  if (nama === 'monitoring') {
    muatMonitoringKegiatanDashboardPTLK();
  }

  if (nama === 'tindakLanjut') {
    muatPilihanMonitoringTindakLanjutPTLK();
    muatTindakLanjutDashboardPTLK();
  }

  if (nama === 'evaluasi') {
    muatEvaluasiKegiatanDashboardPTLK();
  }

  if (nama === 'dokumentasi') {
    muatDokumentasiKegiatanDashboardPTLK();
  }
}

/* =========================================================
   MODUL SIMPAN DAN TAMPILKAN MONEV KEGIATAN PTLK
   ========================================================= */

function PTLK_MONEV_STATUS_(id, pesan) {
  const area = el(id);
  if (area) area.textContent = pesan;
}


function PTLK_MONEV_TABEL_(id, kolom, data) {
  const area = el(id);
  if (!area) return;

  if (!data || !data.length) {
    area.innerHTML = '<p>Belum ada data.</p>';
    return;
  }

  let html = '<div style="overflow-x:auto"><table class="table">';
  html += '<thead><tr>';

  kolom.forEach(function(k) {
    html += '<th>' + PTLK_MONEV_ESCAPE_HTML_(k.label) + '</th>';
  });

  html += '</tr></thead><tbody>';

  data.forEach(function(item) {
    html += '<tr>';

    kolom.forEach(function(k) {
      html += '<td>' +
        PTLK_MONEV_ESCAPE_HTML_(item[k.field]) +
        '</td>';
    });

    html += '</tr>';
  });

  html += '</tbody></table></div>';

  area.innerHTML = html;
}

function simpanDokumentasiKegiatanDashboardPTLK() {
  const idKegiatan = PTLK_MONEV_ID_KEGIATAN_TERPILIH_();

  if (!idKegiatan) {
    alert('Pilih kegiatan terlebih dahulu.');
    return;
  }

  const data = {
    ID_KEGIATAN: idKegiatan,
    JENIS_DOKUMENTASI: el('docJenis').value,
    URAIAN: el('docUraian').value.trim(),
    LINK_BUKTI: el('docTautan').value.trim()
  };

  if (!data.JENIS_DOKUMENTASI || !data.LINK_BUKTI) {
    alert('Jenis dokumentasi dan tautan file wajib diisi.');
    return;
  }

  PTLK_MONEV_STATUS_(
    'statusMonevDokumentasi',
    'Sedang menyimpan dokumentasi...'
  );

  callServer(
    'simpanDokumentasiKegiatanPTLK',
    [TOKEN_LOGIN, data],

    function(hasil) {
      if (!hasil || !hasil.ok) {
        PTLK_MONEV_STATUS_(
          'statusMonevDokumentasi',
          hasil && hasil.pesan ? hasil.pesan : 'Gagal menyimpan dokumentasi.'
        );
        return;
      }

      PTLK_MONEV_STATUS_(
        'statusMonevDokumentasi',
        hasil.pesan + ' ID: ' + hasil.id
      );

      el('docUraian').value = '';
      el('docTautan').value = '';

      muatDokumentasiKegiatanDashboardPTLK();
    },

    function(error) {
      PTLK_MONEV_STATUS_(
        'statusMonevDokumentasi',
        'Gagal menyimpan: ' + error
      );
    }
  );
}

   ===================================================== */

function muatPilihanKegiatanPelaksanaanPTLK() {
  const select = el('pelaksanaanIdKegiatan');
  if (!select) return;

  select.innerHTML = '<option value="">Memuat kegiatan...</option>';

  callServer(
    'ambilDaftarKegiatanDisetujuiPTLK',
    [TOKEN_LOGIN, {}],
    function(hasil) {
      if (!hasil || !hasil.ok) {
        select.innerHTML =
          '<option value="">Gagal memuat kegiatan</option>';
        return;
      }

      select.innerHTML =
        '<option value="">-- Pilih Kegiatan --</option>';

      (hasil.data || []).forEach(function(item) {
        const option = document.createElement('option');
        option.value = item.idKegiatan;
        option.textContent =
          item.tahun + ' | ' +
          item.namaKegiatan + ' | ' +
          item.kabupatenKota;
        select.appendChild(option);
      });

      if (!hasil.data || !hasil.data.length) {
        select.innerHTML =
          '<option value="">Belum ada kegiatan disetujui</option>';
      }
    },
    function(error) {
      select.innerHTML =
        '<option value="">Gagal memuat kegiatan</option>';
      console.error(error);
    }
  );
}


function simpanRealisasiDashboardPTLK() {
  const idKegiatan = el('pelaksanaanIdKegiatan').value;

  if (!idKegiatan) {
    alert('Pilih kegiatan terlebih dahulu.');
    return;
  }

  const data = {
    idKegiatan: idKegiatan,
    outputRealisasi:
      Number(el('pelaksanaanOutput').value || 0),
    anggaranRealisasi:
      Number(el('pelaksanaanAnggaran').value || 0),
    persentaseFisik:
      Number(el('pelaksanaanFisik').value || 0),
    persentaseKeuangan:
      Number(el('pelaksanaanKeuangan').value || 0),
    statusPelaksanaan:
      el('pelaksanaanStatus').value,
    tautanBukti:
      el('pelaksanaanBukti').value.trim(),
    tanggalLaporan:
      el('pelaksanaanTanggal').value,
    catatan:
      el('pelaksanaanCatatan').value.trim()
  };

  const status = el('statusPelaksanaanRealisasi');
  status.textContent = 'Sedang menyimpan realisasi...';

  callServer(
    'simpanRealisasiKegiatanPTLK',
    [TOKEN_LOGIN, data],
    function(hasil) {
      status.textContent =
        hasil && hasil.ok
          ? 'Realisasi berhasil disimpan.'
          : 'Penyimpanan gagal: ' +
            (hasil && hasil.pesan
              ? hasil.pesan
              : JSON.stringify(hasil));

      if (hasil && hasil.ok) {
        muatRealisasiDashboardPTLK();
      }
    },
    function(error) {
      status.textContent =
        'Gagal menyimpan realisasi: ' + error;
    }
  );
}


function muatRealisasiDashboardPTLK() {
  const idKegiatan = el('pelaksanaanIdKegiatan').value;

  if (!idKegiatan) {
    alert('Pilih kegiatan terlebih dahulu.');
    return;
  }

  const area = el('hasilPembacaanRealisasi');
  const status = el('statusPelaksanaanRealisasi');

  area.textContent = 'Sedang membaca data realisasi...';

  callServer(
    'ambilRealisasiKegiatanPTLK',
    [TOKEN_LOGIN, idKegiatan],
    function(hasil) {
      area.textContent = JSON.stringify(hasil, null, 2);

      if (status) {
        status.textContent =
          hasil && hasil.ok
            ? 'Data realisasi berhasil dibaca.'
            : 'Pembacaan realisasi gagal.';
      }

      if (hasil && hasil.ok && hasil.data) {
        const d = hasil.data;

        el('pelaksanaanOutput').value =
          d.outputRealisasi ?? '';

        el('pelaksanaanAnggaran').value =
          d.anggaranRealisasi ?? '';

        el('pelaksanaanFisik').value =
          d.persentaseFisik ?? '';

        el('pelaksanaanKeuangan').value =
          d.persentaseKeuangan ?? '';

        el('pelaksanaanStatus').value =
          d.statusPelaksanaan || 'BELUM DIMULAI';

        el('pelaksanaanTanggal').value =
          d.tanggalLaporan || '';

        el('pelaksanaanBukti').value =
          d.tautanBukti || '';

        el('pelaksanaanCatatan').value =
          d.catatan || '';
      }
    },
    function(error) {
      area.textContent =
        'GAGAL MEMBACA REALISASI:\n' + error;
    }
  );
}

function aturTampilanBerdasarkanPeranPTLK() {

    'panelUjiKoneksiPerencanaanPTLK',
    'panelPelaksanaanRealisasiPTLK',
    'panelDaftarSeluruhRencana',
    'panelPersetujuanRencana',
    'panelRencanaDitolak',
    'panelOperator',
    'panelKelolaOperator'
  ];

  panelTerlarang.forEach(function(id) {
    var panel = document.getElementById(id);
    if (panel) panel.style.display = 'none';
  });

  var judulTerlarang = [
    'Ringkasan Data',
    'Persebaran Data per Kabupaten/Kota',
    'Rekapitulasi 10 OPK',
    'Status Verifikasi',
    'Rincian Data Hasil Filter'
  ];

  document.querySelectorAll('section.panel').forEach(function(panel) {
    var h2 = panel.querySelector('h2');
    if (h2 && judulTerlarang.indexOf(h2.textContent.trim()) !== -1) {
      panel.style.display = 'none';
    }
  });
}
